    'use strict';
import { SendMessageCommand, SQSClient } from "@aws-sdk/client-sqs";
// Initializing the S3 client outside the handler allows connection reuse across invocations.
const sqsClient = new SQSClient({});
const SQS_QUEUE_URL = "https://sqs.us-east-2.amazonaws.com/859196001802/my-simple-standard-queue";


export const handler = async(event) => {
  console.log(event);
  // Parse the input for the name, company property values
  let name = event.name === null || event.name === undefined || event.name === "" ? 'John Doe' : event.name;
  let company = event.company === null || event.company === undefined || event.company === "" ? 'John Doe Company' : event.company;
  let lead_data = name + ', ' + company;
  
//   // Log the lead data to CloudWatch
  console.log('lead data: ', lead_data);


//   // Define the content payload you want to write
    const payload = {
        message: lead_data,
        timestamp: new Date().toISOString(),
    };

    try {
            const messageBody = {
                status: "processed",
                data: payload
            };

            const command = new SendMessageCommand({
                QueueUrl: SQS_QUEUE_URL,
                MessageBody: JSON.stringify(messageBody),
            });

            const response = await sqsClient.send(command);

            return {
                statusCode: 200,
                body: JSON.stringify({
                    message: "Message sent successfully",
                    messageId: response.MessageId,
                }),
            };
            
        } catch (error) {
            console.error(error);
            return {
                statusCode: 500,
                body: JSON.stringify({ error: "Failed to send message" }),
            };
        }

  // Return a greeting to the caller
  //return {"lead data": lead_data}
};

Story-6_Create a pipeline in Databricks that creates a Lead Delta Table.

  • Setup: Upload the brz-lead_full.csv file to the Databricks file system. Upload the delta files to the corresponding subfolder.

  • Lead Pipeline in Databricks

    
%sql
USE CATALOG dbacademy;
USE SCHEMA get_started_de;
SELECT current_catalog(), current_schema();
    
%sql
DROP TABLE IF EXISTS Lead;
    
%sql
SHOW TABLES;
    
%sql
SHOW VOLUMES;
    
%sql
LIST '/Volumes/dbacademy/get_started_de/myfiles/lead';
    
%sql
SELECT * FROM read_files('/Volumes/dbacademy/get_started_de/myfiles/lead/full/brz-lead_full.csv');
    
%sql
SELECT * FROM read_files('/Volumes/dbacademy/get_started_de/myfiles/lead/delta');
    
%sql
CREATE TABLE IF NOT EXISTS lead
AS SELECT * FROM read_files('/Volumes/dbacademy/get_started_de/myfiles/lead/full/brz-lead_full.csv');
    
%sql
DESCRIBE HISTORY lead;
    Version 0 is created with 22 records from the full load file.
    
%sql
SELECT * FROM lead;
    
result = spark.sql(f"""
    COPY INTO lead
    FROM '/Volumes/dbacademy/get_started_de/myfiles/lead/delta'
    FILEFORMAT = CSV
    FORMAT_OPTIONS ('header' = 'true', 'inferSchema' = 'true')
""")
result.display()
    
%sql
SELECT * FROM lead;
    With 5 more lead records from the delta files, the lead table should have 27 records now.
    
%sql
DESCRIBE HISTORY lead;
    Version 0 is created with 27 records from the full load and delta files. The history shows 2 recoords, one from CREATE TABLE AS SELECT operation and one from COPY INTO operation.
    
result = spark.sql(f"""
    COPY INTO lead
    FROM '/Volumes/dbacademy/get_started_de/myfiles/lead/delta'
    FILEFORMAT = CSV
    FORMAT_OPTIONS ('header' = 'true', 'inferSchema' = 'true')
""")
result.display()
     As no new delta files are added, the COPY INTO operation shouldn't affect any rows and no new versions will be created.
    
%sql
DESCRIBE HISTORY lead;
    The history shows same 2 recoords.
