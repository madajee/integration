    touch index-sqs.mjs
    'use strict';
import { SendMessageCommand, SQSClient } from "@aws-sdk/client-sqs";
// Initializing the S3 client outside the handler allows connection reuse across invocations.
const sqsClient = new SQSClient({});
const SQS_QUEUE_URL = "https://sqs.us-east-2.amazonaws.com/859196001802/my-simple-standard-queue";


export const handler = async(event) => {
  console.log(event);
  // Parse the input for the name, company property values
  let name = event.name === null || event.name === undefined || event.name === "" ? 'John Doe' : event.name;
  let company = event.company === null || event.company === undefined || event.company === "" ? 'John Doe Company' : event.company;
  let lead_data = name + ', ' + company;
  
//   // Log the lead data to CloudWatch
  console.log('lead data: ', lead_data);


//   // Define the content payload you want to write
    const payload = {
        message: lead_data,
        timestamp: new Date().toISOString(),
    };

    try {
            const messageBody = {
                status: "processed",
                data: payload
            };

            const command = new SendMessageCommand({
                QueueUrl: SQS_QUEUE_URL,
                MessageBody: JSON.stringify(messageBody),
            });

            const response = await sqsClient.send(command);

            return {
                statusCode: 200,
                body: JSON.stringify({
                    message: "Message sent successfully",
                    messageId: response.MessageId,
                }),
            };
            
        } catch (error) {
            console.error(error);
            return {
                statusCode: 500,
                body: JSON.stringify({ error: "Failed to send message" }),
            };
        }

  // Return a greeting to the caller
  //return {"lead data": lead_data}
};
