  let time = times.indexOf(event.time)<0 ? 'day' : event.time;
  let day = days.indexOf(event.day)<0 ? null : event.day;

  // Generate a greeting
  let greeting = 'Good ' + time + ', ' + name + ' of ' + city + '. ';
  if (day) greeting += 'Happy ' + day + '!';
  
  // Log the greeting to CloudWatch
  console.log('Hello: ', greeting);
  
  // Return a greeting to the caller
  return {"greeting": greeting}
};

  let name = event.name === null || event.name === undefined || event.name === "" ? 'John Doe' : event.name;

  // Generate a greeting
  
  // Log the lead data to CloudWatch
  console.log('lead data: ', lead_data);

  const bucketName = "datapipe-test-bucket-1";
  const fileKey = "data/example-file.json";

  // Define the content payload you want to write
    const payload = {
        message: lead_data,
        timestamp: new Date().toISOString(),
    };

    const params = {
        Bucket: bucketName,
        Key: fileKey,
        Body: JSON.stringify(payload),
        ContentType: "application/json"
    };

    try {
        // Execute the upload via SDK v3 commands
        const response = await s3Client.send(new PutObjectCommand(params));

            statusCode: 500,
            body: JSON.stringify({ message: "Failed to upload data to S3", error: error.message }),
        };
    }

  // Return a greeting to the caller
  return {"lead data": lead_data}
};
